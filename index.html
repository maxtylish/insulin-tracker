☁️ 雲端同步設定

跨裝置同步說明
請先到 supabase.com 建立免費專案，在 SQL Editor 執行以下建表指令，再填入下方資訊。
📋 點此展開建表 SQL（複製貼到 Supabase SQL Editor 執行） 
-- 胰島素紀錄表
create table if not exists insulin_records (
  user_id    text        not null,
  date       date        not null,
  data       jsonb,
  updated_at timestamptz default now(),
  primary key (user_id, date)
);

-- 待辦事項同步（跨裝置共用）
create table if not exists user_todos (
  user_id    text        primary key,
  todos      jsonb       default '[]'::jsonb,
  updated_at timestamptz default now()
);

-- 啟用 Row Level Security（建議）
alter table insulin_records enable row level security;
alter table user_todos enable row level security;
-- 若使用 anon key 匿名存取，請加：
-- create policy "allow all" on insulin_records for all using (true) with check (true);
-- create policy "allow all" on user_todos      for all using (true) with check (true);
