create table if not exists insulin_records (
  user_id    text        not null,
  date       date        not null,
  data       jsonb,
  updated_at timestamptz default now(),
  primary key (user_id, date)
);

-- 待辦事項同步（跨裝置共用）

  user_id    text        primary key,
